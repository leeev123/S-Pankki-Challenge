def _extract_euro_amount(text: str, default: float = 1200.0) -> float:
    """Poimii käyttäjän viestistä ensimmäisen euromäärän (esim. '1 200 €' -> 1200.0)."""
    cleaned = text.replace(" ", "")
    match = re.search(r"(\d+(?:[.,]\d+)?)\s*(?:€|eur|euroa)?", cleaned, re.IGNORECASE)
    if match:
        return float(match.group(1).replace(",", "."))
    return default


def _python_unexpected_expense(customer_id: str, description: str, price: float) -> dict[str, Any]:
    """Deterministinen Python-laskelma yllättävälle menolle."""
    snap = tool_get_financial_summary(customer_id)
    current_buffer = float(snap["profile"].get("current_buffer_eur", 850.0))
    weekly_drop = round(price / 8.0, 2)
    monthly_drop = round(price / 2.0, 2)
    buffer_after = round(max(0.0, current_buffer - price), 2)

    card = {
        "plan_id": f"buffer_absorb_{int(price)}",
        "title": "Kata puskurista & tasaa 2 kk aikana",
        "summary": (
            f"Käytät puskuria {price:.0f} € ({description}) ja vähennät arjen käyttövaraa "
            f"{weekly_drop:.0f} €/vk seuraavat 2 kuukautta puskurin täydentämiseksi."
        ),
        "impact_weekly_eur": -weekly_drop,
        "impact_savings_months": 2,
        "monthly_savings_delta_eur": 0.0,
        "everyday_budget_delta_eur": -monthly_drop,
        "buffer_delta_eur": -price,
        "goal_name": "Puskurin täydennys",
        "goal_target_eur": price,
    }
    return {
        "description": description,
        "price_eur": price,
        "current_buffer_eur": current_buffer,
        "buffer_after_eur": buffer_after,
        "weekly_recovery_drop_eur": weekly_drop,
        "action_card": card,
    }


def _python_category_cut_card(res: dict[str, Any], pct: float) -> dict[str, Any]:
    """Rakentaa toimintokortin tool_simulate_category_cut-tuloksesta Pythonissa."""
    cat = res["category"]
    return {
        "plan_id": f"cut_{cat}_{int(pct)}",
        "title": f"Lukitse +{res['monthly_savings_eur']:.0f} €/kk lisäsäästö",
        "summary": f"Siirretään {res['monthly_savings_eur']:.0f} €/kk säästöön vähentämällä kategoriaa {cat}.",
        "impact_weekly_eur": -res["weekly_gain_eur"],
        "impact_savings_months": None,
        "monthly_savings_delta_eur": res["monthly_savings_eur"],
        "everyday_budget_delta_eur": -res["monthly_savings_eur"],
    }


def _build_gemini_contents(history: list[dict[str, Any]], user_message: str) -> list[types.Content]:
    """Muuntaa Flask-chatin historian google-genai Content-listaksi."""
    contents: list[types.Content] = []
    recent = history[-12:] if history else []

    for msg in recent:
        raw_role = (msg.get("role") or "user").lower()
        role = "model" if raw_role in ("assistant", "model") else "user"
        text = str(msg.get("content") or msg.get("text") or "").strip()
        if text:
            contents.append(types.Content(role=role, parts=[types.Part.from_text(text=text)]))

    # Lisää uusi viesti, ellei kutsuja ole jo lisännyt sitä historyn viimeiseksi
    if not contents or contents[-1].role != "user" or contents[-1].parts[0].text != user_message:
        contents.append(types.Content(role="user", parts=[types.Part.from_text(text=user_message)]))

    return contents


def _run_gemini_agent(customer_id: str, user_message: str, history: list[dict[str, Any]]) -> dict[str, Any]:
    api_key = os.getenv("GEMINI_API_KEY")
    if not api_key:
        raise RuntimeError("GEMINI_API_KEY puuttuu")

    client = genai.Client(api_key=api_key)
    contents = _build_gemini_contents(history, user_message)
    system_instruction = f"{SYSTEM_PROMPT.strip()}\n\n{AGENT_GUARDRAILS}"

    action_cards: list[dict[str, Any]] = []
    final_reply = ""

    for _step in range(5):
        response = client.models.generate_content(
            model=GEMINI_MODEL,
            contents=contents,
            config=types.GenerateContentConfig(
                system_instruction=system_instruction,
                tools=[types.Tool(function_declarations=TOOL_DECLARATIONS)],
                thinking_config=types.ThinkingConfig(thinking_level="LOW"),
            ),
        )

        if not response.function_calls:
            final_reply = (response.text or "").strip()
            break

        # Säilytä mallin vastaus (sisältää toolCall- ja thought_signature-osat)
        if response.candidates and response.candidates[0].content:
            contents.append(response.candidates[0].content)

        function_response_parts: list[types.Part] = []

        for fc in response.function_calls:
            name = fc.name
            args = dict(fc.args or {})

            if name == "get_financial_summary":
                result = tool_get_financial_summary(customer_id)

            elif name == "simulate_purchase":
                item_name = str(args.get("item_name") or "hankinta")
                price = float(args.get("price") or 0.0)
                result = tool_simulate_purchase(customer_id, item_name, price)
                if isinstance(result, dict) and result.get("options"):
                    action_cards = list(result["options"])

            elif name == "simulate_category_cut":
                category = str(args.get("category") or "restaurant")
                pct = max(1.0, min(90.0, float(args.get("reduction_pct") or 20.0)))
                result = tool_simulate_category_cut(customer_id, category, pct)
                action_cards = [_python_category_cut_card(result, pct)]

            elif name == "simulate_unexpected_expense":
                desc = str(args.get("description") or "yllättävä meno")
                price = float(args.get("price") or 0.0)
                result = _python_unexpected_expense(customer_id, desc, price)
                action_cards = [result["action_card"]]

            else:
                result = {"error": f"Tuntematon työkalu: {name}"}

            function_response_parts.append(
                types.Part.from_function_response(
                    name=name,
                    response={"result": result},
                )
            )

        contents.append(types.Content(role="user", parts=function_response_parts))

    if not final_reply:
        final_reply = "Laskin ajantasaiset lukusi. Mitä vaihtoehtoa haluat tarkastella tarkemmin?"

    return {"reply": final_reply, "action_cards": action_cards}


def _run_fallback_agent(customer_id: str, user_message: str) -> dict[str, Any]:
    """Alkuperäinen varalogiikka tilanteisiin, joissa API-avainta ei ole."""
    snap = tool_get_financial_summary(customer_id)
    msg = user_message.lower()

    if any(w in msg for w in ("ostaa", "läppäri", "hankinta", "puhelin", "pyörä", "matka", "€")):
        price = _extract_euro_amount(user_message, 1200.0)
        item = "läppäri" if "läppär" in msg else "suurempi hankinta"
        if "matka" in msg or "loma" in msg:
            item = "lomamatka"
        elif "puhelin" in msg:
            item = "puhelin"

        if any(w in msg for w in ("yllättävä", "lasku", "rikki", "huolto", "pesukone")):
            exp = _python_unexpected_expense(customer_id, item, price)
            reply = (
                f"Ymmärrän — {price:.0f} € yllättävä meno kuuluu **välttämättömiin epäsäännöllisiin menoihin**.\n\n"
                f"Sinulla on tällä hetkellä {exp['current_buffer_eur']:.0f} € puskuri. "
                f"Näin sopeutamme suunnitelmaasi ilman että arki menee sekaisin:"
            )
            return {"reply": reply, "action_cards": [exp["action_card"]]}

        sim = tool_simulate_purchase(customer_id, item, price)
        reply = (
            f"Analysoin hankinnan **{item} ({price:.0f} €)** suhteessa S-Tilisi saldoon "
            f"({snap['account_balance_eur']:.2f} €) ja tämän viikon käyttövaraasi "
            f"({snap['weekly_safe_to_spend_eur']:.2f} €).\n\n"
            f"Koska arvostat asioita **{', '.join(snap['profile'].get('priorities', []))}**, "
            f"en leikkaa niistä. Valitse alta sinulle sopivin tapa:"
        )
        return {"reply": reply, "action_cards": sim["options"]}

    if any(w in msg for w in ("vähen", "säästä", "20%", "ravintol", "kahvi", "taks")):
        pct = _extract_euro_amount(user_message, 20.0)
        if pct > 90:
            pct = 20.0
        cat = "cafe" if "kahv" in msg else ("taxi" if "taks" in msg else "restaurant")
        res = tool_simulate_category_cut(customer_id, cat, pct)

        reply = (
            f"Hyvä laskelma! Viimeisen 30 päivän aikana kategoriaan **{res['category']}** "
            f"meni **{res['current_monthly_eur']:.2f} €**.\n\n"
            f"Jos vähennät sitä **{pct:.0f} %**, vapautat **{res['monthly_savings_eur']:.2f} €/kk** "
            f"(**{res['yearly_savings_eur']:.0f} €/vuosi**) eli vaihtoehtokustannuksena saat "
            f"+{res['weekly_gain_eur']:.2f} €/vk lisää liikkumavaraa muihin tavoitteisiisi."
        )
        return {"reply": reply, "action_cards": [_python_category_cut_card(res, pct)]}

    if any(w in msg for w in ("millainen kuluttaja", "minne rahani", "korit", "analysoi", "profiili")):
        b = snap["baskets_30d"]
        cats = snap["everyday_categories_30d"]
        top_cat = next(iter(cats.items()), ("ruoka", 0))
        reply = (
            f"Viimeisen 30 päivän tilitapahtumasi jakautuvat kolmeen koriin:\n"
            f"1. **Välttämättömät säännölliset:** {b['mandatory_regular_eur']:.0f} € (vuokra, sähkö, vakuutus)\n"
            f"2. **Välttämättömät epäsäännölliset:** {b['mandatory_irregular_eur']:.0f} € (terveys, apteekki, korjaukset)\n"
            f"3. **Arjen kulutus:** {b['everyday_spent_eur']:.0f} € — suurin menoerä on **{top_cat[0]} ({top_cat[1]:.0f} €)**.\n\n"
            f"Kun kiinteät kulut ja {b['monthly_savings_target_eur']:.0f} € säästö on huomioitu, "
            f"tämän viikon käyttövarasi on **{snap['weekly_safe_to_spend_eur']:.2f} €**."
        )
        return {"reply": reply, "action_cards": []}

    reply = (
        f"Tämän viikon jäljellä oleva käyttövarasi on **{snap['weekly_safe_to_spend_eur']:.2f} €** "
        f"(noin **{snap['daily_safe_to_spend_eur']:.2f} €/pv**).\n\n"
        f"Kokeile kysyä esimerkiksi:\n"
        f"• *\"Haluan ostaa 1 200 € läppärin\"*\n"
        f"• *\"Millainen kuluttaja olen?\"*\n"
        f"• *\"Paljonko säästän jos vähennän ravintoloita 20 %?\"*"
    )
    return {"reply": reply, "action_cards": []}


def run_agent(customer_id: str, user_message: str, history: list[dict[str, Any]]) -> dict[str, Any]:
    """
    Kutsutaan aina kun käyttäjä lähettää viestin chatissa.
    Palauttaa tekstivastauksen ja klikattavat toimintokortit.
    """
    if os.getenv("GEMINI_API_KEY"):
        try:
            return _run_gemini_agent(customer_id, user_message, history)
        except Exception as exc:
            logger.warning("Gemini-kutsu epäonnistui, käytetään varalogiikkaa: %s", exc)

    return _run_fallback_agent(customer_id, user_message)

def tool_simulate_category_cut(customer_id: str, category_name: str, cut_percent: float) -> dict:
    """Laskee kuukausi- ja vuosisäästön, jos tiettyä kulukategoriaa vähennetään X %."""
    snap = get_financial_snapshot(customer_id)
    cats = snap["everyday_categories_30d"]

    matched_val = 0.0
    matched_key = category_name
    for k, v in cats.items():
        if category_name.lower() in k.lower() or k.lower() in category_name.lower():
            matched_val += v
            matched_key = k

    if matched_val == 0:
        matched_val = 120.0

    monthly_save = round(matched_val * (cut_percent / 100.0), 2)
    return {
        "category": matched_key,
        "current_monthly_eur": matched_val,
        "cut_percent": cut_percent,
        "monthly_savings_eur": monthly_save,
        "yearly_savings_eur": round(monthly_save * 12, 2),
        "weekly_gain_eur": round(monthly_save / 4.33, 2),
    }

def _load_db() -> dict[str, Any]:
    if not DB_PATH.exists():
        return {"profiles": {}, "histories": {}}
    try:
        return json.loads(DB_PATH.read_text(encoding="utf-8"))
    except json.JSONDecodeError:
        return {"profiles": {}, "histories": {}}


def _save_db(data: dict[str, Any]) -> None:
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    DB_PATH.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def get_profile(customer_id: str) -> dict[str, Any]:
    """Hakee profiilin tai alustaa sen asiakasluokan (Onni, Enni, Anni, Oma) mukaan."""
    db = _load_db()
    if customer_id in db["profiles"]:
        return db["profiles"][customer_id]

    preset = PERSONA_DEFAULTS.get(customer_id, {
        "display_name": "Oma tiliote",

    })

    return {
        "customer_id": customer_id,
        "s_ai_enabled": False,  # Alkaa aina perinteisenä S-mobiilina ennen aktivointia
        "onboarded": False,
        "everyday_adjustment_eur": 0.0,
        "active_goal": None,
        **preset,

def save_profile(customer_id: str, profile: dict[str, Any]) -> dict[str, Any]:
    db = _load_db()
    db["profiles"][customer_id] = profile
    _save_db(db)
    return profile


def get_history(customer_id: str) -> list[dict[str, Any]]:
    db = _load_db()
    return db["histories"].get(customer_id, [])

    _save_db(db)

def _resolve_csv_paths() -> tuple[Path, Path]:
    ensure_sample_dataset()
    tx_env = Path(os.getenv("SBANK_TX_CSV", str(DATA_DIR / "tx_junction_transactions.csv")))
    cust_env = Path(os.getenv("SBANK_CUSTOMERS_CSV", str(DATA_DIR / "tx_customers_junction.csv")))
    if not tx_env.is_absolute():
        tx_env = Path(__file__).resolve().parents[1] / tx_env
    if not cust_env.is_absolute():
        cust_env = Path(__file__).resolve().parents[1] / cust_env

    tx_path = tx_env if tx_env.exists() else SAMPLE_TX_PATH
    cust_path = cust_env if cust_env.exists() else SAMPLE_CUST_PATH
    return tx_path, cust_path


@lru_cache(maxsize=8)

    _, cust_path = _resolve_csv_paths()
    df = pd.read_csv(cust_path, nrows=limit)
    return df.to_dict(orient="records") # type: ignore


@lru_cache(maxsize=16)
def load_customer_transactions(customer_id: str) -> pd.DataFrame:
    """
    Hakee asiakkaan tapahtumat:
    - Jos valittuna on oma ladattu Excel (uploaded_<customer_id>.xlsx), luetaan se.
    - Muuten luetaan S-Pankin CSV (tai valmiit Onni/Enni/Anni-profiilit).
    """
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    custom_excel = DATA_DIR / f"uploaded_{customer_id}.xlsx"

    if custom_excel.exists():
        df = parse_excel_statement(custom_excel, customer_id=customer_id)
        return df

    # Jos kansiossa on muu yksittäinen .xlsx ja asiakas on CUST_CUSTOM
    if customer_id == "CUST_CUSTOM":
        other_excels = [p for p in sorted(DATA_DIR.glob("*.xlsx")) if not p.name.startswith("uploaded_")]
        if other_excels:
            return parse_excel_statement(other_excels[0], customer_id=customer_id)

    tx_path, _ = _resolve_csv_paths()
    chunks = []
    for chunk in pd.read_csv(tx_path, chunksize=100_000):
        matched = chunk[chunk["customer_id"].astype(str) == str(customer_id)]
        if not matched.empty:
            chunks.append(matched)

    if not chunks:
        first_chunk = pd.read_csv(tx_path, nrows=2000)
        first_id = str(first_chunk["customer_id"].iloc[0])
        df = first_chunk[first_chunk["customer_id"].astype(str) == first_id].copy()
    else:
        df = pd.concat(chunks, ignore_index=True)

    df["tx_date"] = pd.to_datetime(df["tx_date"], errors="coerce")
    df["tx_amount_eur"] = pd.to_numeric(df["tx_amount_eur"], errors="coerce").fillna(0.0)
    return df.sort_values("tx_date", ascending=False)


def classify_basket(category: str, amount: float) -> str:
    if amount > 0:
        return "income"
    cat = str(category).strip().lower()
    if any(k in cat for k in MANDATORY_REGULAR_KEYS):
        return "mandatory_regular"
    if any(k in cat for k in MANDATORY_IRREGULAR_KEYS):
        return "mandatory_irregular"
    return "everyday_discretionary"


def get_financial_snapshot(customer_id: str = "CUST_ANNI") -> dict[str, Any]:
    df = load_customer_transactions(customer_id).copy()
    profile = get_profile(customer_id)

    df["basket"] = [
        classify_basket(cat, amt) for cat, amt in zip(df["category"], df["tx_amount_eur"])
    ]

    max_date = df["tx_date"].max()
    last_30 = df[df["tx_date"] >= (max_date - pd.Timedelta(days=30))]
    last_7 = df[df["tx_date"] >= (max_date - pd.Timedelta(days=7))]

    monthly_income = float(last_30[last_30["tx_amount_eur"] > 0]["tx_amount_eur"].sum())
    if monthly_income <= 0:
        monthly_income = 2500.0

    mandatory_regular = abs(float(last_30[last_30["basket"] == "mandatory_regular"]["tx_amount_eur"].sum()))
    mandatory_irregular = abs(float(last_30[last_30["basket"] == "mandatory_irregular"]["tx_amount_eur"].sum()))
    everyday_30d = abs(float(last_30[last_30["basket"] == "everyday_discretionary"]["tx_amount_eur"].sum()))
    spent_this_week = abs(float(last_7[last_7["basket"] == "everyday_discretionary"]["tx_amount_eur"].sum()))

    savings_target = float(profile.get("monthly_savings_target_eur", 250.0))
    everyday_adj = float(profile.get("everyday_adjustment_eur", 0.0))

    irregular_reserve = max(60.0, round(mandatory_irregular * 1.1, 2))
    monthly_everyday_budget = max(
        180.0,
        round(monthly_income - mandatory_regular - irregular_reserve - savings_target + everyday_adj, 2),
    )
    weekly_budget = round(monthly_everyday_budget / 4.33, 2)
    weekly_safe_to_spend = round(weekly_budget - spent_this_week, 2)
    daily_safe_to_spend = round(max(0.0, weekly_safe_to_spend) / 4.0, 2)

    account_balance = float(df["balance_after_eur"].iloc[0]) if "balance_after_eur" in df.columns else 853.34

    everyday_df = last_30[last_30["basket"] == "everyday_discretionary"]
    by_category = (
        everyday_df.groupby("category")["tx_amount_eur"]
        .sum()
        .abs()
        .sort_values(ascending=False)
        .round(2)
        .to_dict()
    )

    recent_receipts = []
    for _, row in df.head(6).iterrows():
        tx_d = row["tx_date"]
        date_label = "Tänään 07:20" if pd.notna(tx_d) and tx_d.date() == max_date.date() else (
            tx_d.strftime("%d.%m.%Y") if pd.notna(tx_d) else ""
        )
        recent_receipts.append({
            "date": date_label,
            "merchant": str(row.get("merchant_id", row.get("category", "Ostos"))),
            "category": str(row.get("category", "")),
            "amount_eur": round(float(row["tx_amount_eur"]), 2),
            "basket": row["basket"],
        })

    return {
        "customer_id": customer_id,
        "account_balance_eur": round(account_balance, 2),
        "monthly_income_eur": round(monthly_income, 2),
        "weekly_budget_eur": weekly_budget,
        "spent_this_week_eur": round(spent_this_week, 2),
        "weekly_safe_to_spend_eur": weekly_safe_to_spend,
        "daily_safe_to_spend_eur": daily_safe_to_spend,
        "baskets_30d": {
            "mandatory_regular_eur": round(mandatory_regular, 2),
            "mandatory_irregular_eur": round(mandatory_irregular, 2),
            "irregular_reserve_eur": round(irregular_reserve, 2),
            "everyday_spent_eur": round(everyday_30d, 2),
            "monthly_everyday_budget_eur": monthly_everyday_budget,
            "monthly_savings_target_eur": round(savings_target, 2),
        },
        "everyday_categories_30d": by_category,
        "recent_transactions": recent_receipts,
        "profile": profile,
    }

# Avainsanat, joilla päätellään kategoria automaattisesti, jos Excel-tiliotteessa ei ole Kategoria-saraketta
KEYWORD_CATEGORY_MAP = {
    "salary": ["palkka", "salary", "etuus", "kela", "opintotuki"],
    "rent": ["vuokra", "asunto", "hoas", "sato", "lumo", "vastike"],

    "telecom": ["elisa", "telia", "dna", "moi"],

    "pharmacy": ["apteekki", "yliopiston apteekki"],
    "health": ["terveystalo", "mehiläinen", "hammas", "lääkäri"],
    "repair": ["autohuolto", "korjaamo", "gigantti", "verkkokauppa"],
    "groceries": ["s-market", "prisma", "alepa", "sale", "k-market", "k-citymarket", "lidl", "ruoka"],

    "restaurant": ["ravintola", "wolt", "foodora", "hesburger", "mcdonald", "friends & brgrs", "pizzeria"],
    "taxi": ["taksi", "taxi", "bolt", "uber", "yang"],
}


def _clean_euro_value(val: Any) -> float:
    """Muuttaa erilaiset euromuodot (esim. '-12,50 €', '1 200.00') Pythonin float-luvuksi."""
    if pd.isna(val):
        return 0.0
    if isinstance(val, (int, float)):
        return float(val)
    text = str(val).replace("€", "").replace("\xa0", "").replace(" ", "").replace(",", ".")
    match = re.search(r"-?\d+(?:\.\d+)?", text)
    return float(match.group(0)) if match else 0.0


def _infer_category(merchant_or_desc: str, amount: float) -> str:
    """Päättelee kategorian saajan/maksajan tai selitteen perusteella."""
    text = str(merchant_or_desc).lower()
    if amount > 0:
        for kw in KEYWORD_CATEGORY_MAP["salary"]:
            if kw in text:
                return "salary"
        return "income"

    for category, keywords in KEYWORD_CATEGORY_MAP.items():
        if category == "salary":
            continue
        if any(kw in text for kw in keywords):
            return category
    return "other_everyday"


def _find_matching_column(columns: list[str], candidates: list[str]) -> str | None:
    """Etsii DataFramesta ensimmäisen sarakkeen, joka vastaa jotain yleistä suomi/englanti-nimeä."""
    lower_map = {str(c).strip().lower(): c for c in columns}
    for cand in candidates:
        if cand in lower_map:
            return lower_map[cand]
    # Osittainen osuma (esim. 'Määrä EUR' sisältää sanan 'määrä')
    for col_lower, orig_col in lower_map.items():
        if any(cand in col_lower for cand in candidates):
            return orig_col
    return None


def parse_excel_statement(file_path: Path, customer_id: str = "CUST_001") -> pd.DataFrame:
    """
    Lukee Excel-tiliotteen (.xlsx / .xls) ja muuntaa sen samaan muotoon
    kuin S-Pankin synteettinen CSV-data.
    """
    # Luetaan Excel-tiedoston ensimmäinen välilehti
    raw_df = pd.read_excel(file_path, engine="openpyxl")
    raw_df = raw_df.dropna(how="all")  # Poistetaan täysin tyhjät rivit

    cols = list(raw_df.columns)

    # Etsitään joustavasti yleisimmät sarakkeet (S-Pankki, Nordea, OP tai oma synteettinen Excel)
    date_col = _find_matching_column(cols, ["tx_date", "kirjauspäivä", "maksupäivä", "päivämäärä", "pvm", "date"])
    amount_col = _find_matching_column(cols, ["tx_amount_eur", "määrä", "summa", "amount", "veloitus/hyvitys"])
    merchant_col = _find_matching_column(cols, ["merchant_id", "saaja/maksaja", "saaja", "maksaja", "selite", "kuvaus", "merchant", "viesti"])
    category_col = _find_matching_column(cols, ["category", "kategoria", "luokka", "tapahtumalaji"])
    balance_col = _find_matching_column(cols, ["balance_after_eur", "saldo", "balance"])
    cust_col = _find_matching_column(cols, ["customer_id", "asiakasnumero"])

    if not amount_col:
        raise ValueError("Excel-tiedostosta ei löytynyt summa-saraketta (esim. 'Summa', 'Määrä' tai 'tx_amount_eur').")

    normalized = pd.DataFrame()
    normalized["customer_id"] = raw_df[cust_col].astype(str) if cust_col else customer_id
    normalized["bank_account_id"] = "EXCEL_IMPORT"

    # Päivämäärän parsiminen (tukee myös suomalaista muotoa dd.mm.yyyy)
    if date_col:
        normalized["tx_date"] = pd.to_datetime(raw_df[date_col], dayfirst=True, errors="coerce")
    else:
        normalized["tx_date"] = pd.Timestamp.today()

    normalized["tx_timestamp"] = normalized["tx_date"].dt.strftime("%Y-%m-%dT12:00:00")
    normalized["tx_amount_eur"] = raw_df[amount_col].apply(_clean_euro_value)

    if merchant_col:
        normalized["merchant_id"] = raw_df[merchant_col].fillna("Tuntematon kauppias").astype(str)
    else:
        normalized["merchant_id"] = "Tiliotetapahtuma"

    # Jos Excelissä on valmis kategoria, käytetään sitä; muuten päätellään
    if category_col:
        normalized["category"] = [
            str(cat).strip() if pd.notna(cat) and str(cat).strip() else _infer_category(merch, amt)
            for cat, merch, amt in zip(raw_df[category_col], normalized["merchant_id"], normalized["tx_amount_eur"])
        ]
    else:
        normalized["category"] = [
            _infer_category(merch, amt)
            for merch, amt in zip(normalized["merchant_id"], normalized["tx_amount_eur"])
        ]

    # Jos saldo puuttuu Excelistä, simuloidaan juokseva saldo järjestyksessä
    if balance_col:
        normalized["balance_after_eur"] = raw_df[balance_col].apply(_clean_euro_value)
    else:
        normalized = normalized.sort_values("tx_date", ascending=True)
        normalized["balance_after_eur"] = (1200.0 + normalized["tx_amount_eur"].cumsum()).round(2)

function renderHistory(history) {
  const el = document.getElementById("chatMessages");
  if (!history.length) {
    el.innerHTML = `<div class="msg assistant">Moikka! Olen Urho, S-AI-talousvalmentajasi. Kysy viikon käyttövarastasi tai kokeile hankinnan suunnittelua (esim. "Haluan ostaa 1 200 € läppärin").</div>`;
    return;
  }
  el.innerHTML = history
    .map((m) => {
      const cardsHtml = (m.action_cards || [])
        .map(
          (c) => `
          <div class="action-card">
            <h4>${c.title}</h4>
            <p>${c.summary}</p>
            <button class="apply-btn" onclick='applyPlan(${JSON.stringify(c)})'>Hyväksy suunnitelma</button>
          </div>`
        )
        .join("");
      return `<div class="msg ${m.role}">${m.content.replace(/\*\*(.*?)\*\*/g, "<strong>$1</strong>")}${cardsHtml}</div>`;
    })
    .join("");
  el.scrollTop = el.scrollHeight;
}

async function sendMessage(text) {
  if (!text.trim()) return;
  const res = await fetch("/api/chat", {

    body: JSON.stringify({ customer_id: currentCustomerId, message: text }),
  });
  const data = await res.json();
  renderSnapshot(data.updated_state);
  await loadState();
}

  });
  const data = await res.json();
  renderSnapshot(data.snapshot);
  renderHistory(data.history);
  switchTab("home");
}

async function uploadExcelFile(targetCustId, file, statusEl) {
  const formData = new FormData();
  formData.append("file", file);
  const res = await fetch(`/api/upload-excel/${targetCustId}`, { method: "POST", body: formData });
  if (!res.ok) {
    const err = await res.json();
    statusEl.textContent = `Virhe: ${err.detail}`;
    return false;
  }
  statusEl.textContent = "Excel-tiliote ladattu!";
  return true;
}
