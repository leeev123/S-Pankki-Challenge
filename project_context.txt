def _discover_models(api_key: str) -> list[str]:
    global _DISCOVERED_MODELS
    if _DISCOVERED_MODELS is not None:
        return _DISCOVERED_MODELS
    url = f"https://generativelanguage.googleapis.com/v1beta/models?key={api_key}"
    found: list[str] = []
    try:
        with urllib.request.urlopen(urllib.request.Request(url, method="GET"), timeout=10) as resp:
            payload = json.loads(resp.read().decode("utf-8"))
        for m in payload.get("models") or []:
            if "generateContent" not in (m.get("supportedGenerationMethods") or []):
                continue
            name = str(m.get("name") or "").replace("models/", "").strip()
            skip_words = ("embedding", "imagen", "veo", "tts", "audio", "vision", "2.5-flash", "1.5")
            if name.startswith("gemini-") and not any(w in name for w in skip_words):
                found.append(name)
        found.sort(key=lambda s: ("3.8" in s, "pro" in s and "preview" not in s, "flash" in s, s), reverse=True)
    except Exception:
        pass
    _DISCOVERED_MODELS = found
    return found

def _to_float(value: Any, default: float = 0.0) -> float:
    if value is None:
        return default
    if isinstance(value, (int, float)):
        return float(value)
    try:
        return float(str(value).replace(" ", "").replace(",", "."))
    except (TypeError, ValueError):
        return default


def _extract_optional_euro(text: str) -> float | None:
    cleaned = text.replace(" ", "")
    match = re.search(r"(\d+(?:[.,]\d+)?)\s*(?:â‚¬|eur|euroa|e\b|tonni|k\b)?", cleaned, re.IGNORECASE)
    if not match:
        return None
    val = _to_float(match.group(1), 0.0)
    if "tonni" in text.lower():
        val *= 1000.0
    return val if val > 0 else None


def _strip_redundant_list_when_cards(text: str, has_cards: bool, lang: str) -> str:
    if not has_cards or not text:
        return text
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
    kept = [ln for ln in lines if not re.match(r"^(\d+[\.\)]|[-â€¢*])\s+", ln)]
    compact = " ".join(kept).strip()
    if not compact:
        return "Choose a suitable option below or adjust the timeline:" if lang == "en" else "Valitse alta sopivin vaihtoehto tai sÃ¤Ã¤dÃ¤ tahtia liukusÃ¤Ã¤timellÃ¤:"
    sentences = re.split(r"(?<=[.!?])\s+", compact)
    return " ".join(sentences[:2]).strip()


def _build_rest_contents(history: list[dict[str, Any]], user_message: str) -> list[dict[str, Any]]:
    contents: list[dict[str, Any]] = []
    for msg in (history[-14:] if history else []):
        role = "model" if str(msg.get("role") or "").lower() in ("assistant", "model") else "user"
        text = str(msg.get("content") or msg.get("text") or "").strip()
        if not text:
            continue
        if contents and contents[-1]["role"] == role:
            if contents[-1]["parts"][0]["text"] != text:
                contents[-1]["parts"][0]["text"] += f"\n{text}"
        else:
            contents.append({"role": role, "parts": [{"text": text}]})
    clean_message = user_message.strip()
    if clean_message:
        if not contents or contents[-1]["role"] != "user":
            contents.append({"role": "user", "parts": [{"text": clean_message}]})
        elif contents[-1]["parts"][0]["text"] != clean_message:
            contents[-1]["parts"][0]["text"] += f"\n{clean_message}"
    return contents


def _execute_tool(statement_id: str, tool_name: str, args: dict[str, Any]) -> tuple[dict[str, Any], list[dict[str, Any]] | None]:
    if tool_name == "get_financial_summary":
        return tool_get_financial_summary(statement_id), None
    if tool_name == "propose_deficit_budget":
        src = str(args.get("funding_source") or "omat varat").strip()
        amt = max(0.0, _to_float(args.get("monthly_funding_eur"), 0.0))
        trim = max(0.0, _to_float(args.get("everyday_trim_eur"), 0.0))
        res = tool_propose_deficit_budget(statement_id, src, amt, trim)
        return res, [res["action_card"]]
    if tool_name == "find_savings_opportunities":
        return tool_find_savings_opportunities(statement_id), None
    if tool_name == "simulate_purchase":
        item_name = str(args.get("item_name") or "").strip()
        price = _to_float(args.get("price"), 0.0)
        housing_words = ("talo", "asunto", "omakotitalo", "kaksio", "yksiÃ¶", "kolmio", "house", "apartment", "flat")
        if price <= 0 or price > 25000 or any(w in item_name.lower() for w in housing_words):
            snap = tool_get_financial_summary(statement_id)
            return {"note": "Ask 1â€“2 clarifying questions about target price range, savings/down payment, and timeline.", "financial_summary": snap}, []
        sim = tool_simulate_purchase(statement_id, item_name or "purchase", price)
        return sim, list(sim.get("options", []))
    if tool_name == "simulate_category_cut":
        category = str(args.get("category") or "ravintolat").strip() or "ravintolat"
        pct = max(1.0, min(90.0, _to_float(args.get("reduction_pct"), 20.0)))
        cut_res = tool_simulate_category_cut(statement_id, category, pct)
        card = cut_res.get("action_card")
        cards = [card] if isinstance(card, dict) and cut_res.get("monthly_savings_eur", 0) > 0 else []
        return cut_res, cards
    if tool_name == "simulate_unexpected_expense":
        desc = str(args.get("description") or "expense").strip() or "expense"
        price = max(0.0, _to_float(args.get("price"), 0.0))
        if price <= 0:
            return {"error": "Ask the user for the expense amount."}, []
        exp_res = tool_simulate_unexpected_expense(statement_id, desc, price)
        card = exp_res.get("action_card")
        return exp_res, ([card] if isinstance(card, dict) else None)
    return {"error": f"Unknown tool: {tool_name}"}, None


def _call_gemini_rest(api_key: str, payload: dict[str, Any]) -> dict[str, Any]:
    global _ACTIVE_MODEL
    body_bytes = json.dumps(payload, ensure_ascii=False).encode("utf-8")
    last_error = ""
    for model_name in _candidate_models(api_key):
        url = f"https://generativelanguage.googleapis.com/v1beta/models/{model_name}:generateContent?key={api_key}"
        req = urllib.request.Request(url, data=body_bytes, headers={"Content-Type": "application/json; charset=utf-8"}, method="POST")
        try:
            with urllib.request.urlopen(req, timeout=28) as resp:
                data = json.loads(resp.read().decode("utf-8"))
            if _ACTIVE_MODEL != model_name:
                _ACTIVE_MODEL = model_name
                logger.warning("Suunta AI kayttaa kielimallia: %s", model_name)
            return data
        except urllib.error.HTTPError as http_exc:
            err_body = http_exc.read().decode("utf-8", errors="replace")
            last_error = f"{model_name} -> HTTP {http_exc.code}: {err_body[:200]}"
            if http_exc.code in (404, 400) and _ACTIVE_MODEL == model_name:
                _ACTIVE_MODEL = None
            continue
        except Exception as exc:
            last_error = f"{model_name} -> {exc}"
            continue
    raise RuntimeError(f"Gemini REST -kutsu epaonnistui: {last_error}")


def _run_gemini_agent(statement_id: str, user_message: str, history: list[dict[str, Any]], lang: str = "fi") -> dict[str, Any]:
    api_key = os.getenv("GEMINI_API_KEY", "").strip()
    if not api_key or api_key in ("MY_GEMINI_API_KEY", "your_api_key_here"):
        raise RuntimeError("GEMINI_API_KEY puuttuu.")

    lang_rule = (
        "\n\nLANGUAGE RULE: The user interface language is ENGLISH. Respond STRICTLY in clear, natural English!"
        if lang == "en"
        else "\n\nKIELISÃ„Ã„NTÃ–: Vastaa selkeÃ¤llÃ¤ ja luonnollisella suomen kielellÃ¤."
    )
    contents = _build_rest_contents(history, user_message)
    system_instruction = f"{SUUNTA_SYSTEM_PROMPT.strip()}\n\n{AGENT_GUARDRAILS}{lang_rule}"
    action_cards: list[dict[str, Any]] = []
    final_reply = ""

    for _step in range(5):
        payload = {
            "systemInstruction": {"parts": [{"text": system_instruction}]},
            "contents": contents,
            "tools": [{"functionDeclarations": TOOL_DECLARATIONS_JSON}],
            "generationConfig": {"temperature": 0.3},
        }
        data = _call_gemini_rest(api_key, payload)
        candidates = data.get("candidates") or []
        if not candidates:
            break
        model_content = candidates[0].get("content") or {}
        parts = model_content.get("parts") or []
        fcalls = [p["functionCall"] for p in parts if isinstance(p, dict) and "functionCall" in p]
        if not fcalls:
            visible_texts = [
                str(p.get("text", "")).strip()
                for p in parts
                if isinstance(p, dict) and not p.get("thought") and str(p.get("text", "")).strip()
            ]
            final_reply = "\n\n".join(visible_texts).strip()
            break

        contents.append(model_content)
        fn_responses: list[dict[str, Any]] = []
        for fc in fcalls:
            tool_name = str(fc.get("name") or "").strip()
            if not tool_name:
                continue
            args = fc.get("args") if isinstance(fc.get("args"), dict) else {}
            result, new_cards = _execute_tool(statement_id, tool_name, args)
            if new_cards is not None:
                action_cards = [_translate_card_en(c) for c in new_cards] if lang == "en" else new_cards
            fn_responses.append({"functionResponse": {"name": tool_name, "response": {"result": result}}})
        if not fn_responses:
            break
        contents.append({"role": "user", "parts": fn_responses})

    final_reply = _strip_redundant_list_when_cards(final_reply, bool(action_cards), lang)
    if not final_reply:
        final_reply = (
            ("Choose an option below:" if action_cards else "Could you tell me a bit more about your goal?")
            if lang == "en"
            else ("Valitse alta sopivin vaihtoehto:" if action_cards else "Kerrotko hieman tarkemmin tilanteestasi?")
        )
    return {"reply": final_reply, "action_cards": action_cards}

def run_agent(statement_id: str, user_message: str, history: list[dict[str, Any]], lang: str = "fi") -> dict[str, Any]:

def _load_db() -> dict[str, Any]:
    if not DB_PATH.exists():
        return {"profiles": {}, "histories": {}}
    try:
        return json.loads(DB_PATH.read_text(encoding="utf-8"))
    except json.JSONDecodeError:
        return {"profiles": {}, "histories": {}}


def _save_db(data: dict[str, Any]) -> None:
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)
    DB_PATH.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")

@lru_cache(maxsize=4)

def _build_financial_engine(df: pd.DataFrame, profile: dict[str, Any]) -> FinancialEngine:
    max_date = df["tx_date"].max().date()
    tx_records: list[dict[str, Any]] = []
    for idx, row in df.iterrows():
        amt_cents = cents(row["tx_amount_eur"])
        desc = str(row.get("merchant_id", ""))
        raw_cat = str(row.get("category", ""))
        tx_records.append({
            "id": f"tx-{idx}",
            "date": row["tx_date"].date(),
            "description": desc,
            "amount_cents": amt_cents,
            "provided_category": raw_cat,
            **categorize(desc, amt_cents, raw_cat),
        })
    return FinancialEngine(tx_records, profile=profile, as_of=max_date)

class FinancialError(ValueError):

def cents(value: Any) -> int:
    """Parse an EUR amount into exact integer cents."""
    if isinstance(value, bool) or value is None or str(value).strip() == "":
        raise FinancialError("RahamÃ¤Ã¤rÃ¤ puuttuu tai on virheellinen")
    raw = str(value).strip().replace("\u00a0", "").replace(" ", "").replace("â‚¬", "")
    if raw.count(",") and raw.count("."):
        decimal_mark = "," if raw.rfind(",") > raw.rfind(".") else "."
        grouping_mark = "." if decimal_mark == "," else ","
        raw = raw.replace(grouping_mark, "").replace(decimal_mark, ".")
    elif "," in raw:
        raw = raw.replace(",", ".")
    try:
        number = Decimal(raw)
        if not number.is_finite():
            raise InvalidOperation
        return int((number * 100).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    except (InvalidOperation, ValueError) as exc:
        raise FinancialError(f"Virheellinen rahamÃ¤Ã¤rÃ¤: {value!r}") from exc


def eur(amount_cents: int) -> str:
    """Format integer cents into '0.00' string."""
    sign = "-" if amount_cents < 0 else ""
    val = abs(amount_cents)
    return f"{sign}{val // 100}.{val % 100:02d}"


def cents_to_eur(amount_cents: int) -> float:
    """Convert integer cents to a clean 2-decimal EUR float for JSON responses."""
    return round(amount_cents / 100.0, 2)


def proportional(value: int, numerator: int, denominator: int) -> int:
    if denominator == 0:
        raise FinancialError("Nollalla ei voi jakaa")
    return int((Decimal(value) * numerator / denominator).quantize(Decimal("1"), rounding=ROUND_HALF_UP))


def monthly_to_weekly(value: int) -> int:
    return proportional(value, 12, 52)


def parse_date(value: Any) -> date:
    if isinstance(value, datetime):
        return value.date()
    if isinstance(value, date):
        return value
    if not value:
        raise FinancialError("Tapahtuman pÃ¤ivÃ¤mÃ¤Ã¤rÃ¤ puuttuu")
    raw = str(value).strip()
    for fmt in ("%Y-%m-%d", "%d.%m.%Y", "%d/%m/%Y", "%Y/%m/%d", "%Y-%m-%d %H:%M:%S", "%Y-%m-%dT%H:%M:%S"):
        try:
            return datetime.strptime(raw[:19], fmt).date()
        except ValueError:
            pass
    raise FinancialError(f"PÃ¤ivÃ¤mÃ¤Ã¤rÃ¤ ei kelpaa: {value!r}")


def add_months(day: date, months: int) -> date:
    new_month_index = day.year * 12 + day.month - 1 + months
    year, remainder = divmod(new_month_index, 12)
    month = remainder + 1
    return date(year, month, min(day.day, calendar.monthrange(year, month)[1]))


def _key(value: str) -> str:
    value = unicodedata.normalize("NFKD", str(value))
    value = "".join(char for char in value if not unicodedata.combining(char))
    return re.sub(r"[^a-z0-9]+", "_", value.casefold()).strip("_")

def resolve_category_alias(raw_category: str) -> str | None:

def category_label_fi(category_key: str) -> str:
    canonical = resolve_category_alias(category_key) or category_key
    return CATEGORY_LABELS_FI.get(canonical, str(category_key).capitalize())


def priorities_to_protected_categories(priorities: list[str] | None) -> list[str]:
    protected: set[str] = set()
    for p in priorities or []:
        norm = _key(p)
        if norm in PRIORITY_TO_CATEGORIES:
            protected |= PRIORITY_TO_CATEGORIES[norm]
        elif "ruoka" in norm:
            protected.add("groceries")
        elif "liikunta" in norm or "harrast" in norm:
            protected.add("leisure")
    return sorted(protected & EVERYDAY)


def categorize(description: str, amount_cents: int, provided_category: str | None = None) -> dict[str, Any]:
    provided = _key(provided_category or "")
    category = CATEGORY_ALIASES.get(provided)
    source = "provided" if category else "unclassified"
    confidence = 1.0 if category else 0.0
    if not category:
        norm_desc = _key(description).replace("_", " ")
        for pattern, candidate, score in MERCHANT_RULES:
            if re.search(pattern, norm_desc):
                category, confidence, source = candidate, score, "merchant_rule"
                break
    if not category:
        category = "other_income" if amount_cents > 0 else "other_everyday"
        confidence = 0.5
        source = "sign_only"
    if category in TRANSFERS:
        bucket = "excluded_transfer"
    elif amount_cents > 0:
        bucket = "income"
        if category not in INCOME:
            category, confidence, source = "other_income", 0.5, "sign_only"
    elif category in INCOME:
        bucket = "income_reversal"
    elif category in REGULAR:
        bucket = "mandatory_regular"
    elif category in IRREGULAR:
        bucket = "mandatory_irregular"
    elif category in EVERYDAY:
        bucket = "everyday"
    else:
        bucket = "unclassified"
    return {
        "category": category,
        "bucket": bucket,
        "classification_confidence": confidence,
        "classification_source": source,
        "needs_review": confidence < 0.75,
    }


class FinancialEngine:
    """Pure deterministic finance calculations powered by integer cents."""

    def __init__(self, transactions: list[dict[str, Any]], profile: dict[str, Any] | None = None, *, as_of: str | date | None = None):
        self.transactions = list(transactions)
        self.profile = dict(profile or {})
        self.as_of = parse_date(as_of) if as_of else date.today()
        self.corrections: dict[str, str] = {}

    def _tx(self, start: date | None = None, end: date | None = None) -> list[dict[str, Any]]:
        output = []
        for tx in self.transactions:
            if tx["date"] > self.as_of or (start and tx["date"] < start) or (end and tx["date"] > end):
                continue
            item = dict(tx)
            if tx["id"] in self.corrections:
                updated = categorize(tx["description"], tx["amount_cents"], self.corrections[tx["id"]])
                item.update(updated)
                item["classification_source"] = "user_correction"
                item["needs_review"] = False
            output.append(item)
        return output

    def analyze_transactions(self, *, start: str | date | None = None, end: str | date | None = None) -> dict[str, Any]:
        start_d = parse_date(start) if start else None
        end_d = parse_date(end) if end else self.as_of
        if start_d and start_d > end_d:
            raise FinancialError("AlkupÃ¤ivÃ¤ on loppupÃ¤ivÃ¤n jÃ¤lkeen")
        transactions = self._tx(start_d, end_d)
        by_cat: dict[str, int] = defaultdict(int)
        by_bucket: dict[str, int] = defaultdict(int)
        income, expenses, transfers, unclassified = 0, 0, 0, 0
        for tx in transactions:
            amount = tx["amount_cents"]
            if tx["bucket"] == "excluded_transfer":
                transfers += abs(amount)
            elif tx["bucket"] == "income_reversal":
                income += amount
            elif amount > 0:
                income += amount
            elif amount < 0:
                expenses += -amount
                by_cat[tx["category"]] += -amount
                by_bucket[tx["bucket"]] += -amount
                if tx["bucket"] == "unclassified":
                    unclassified += -amount
        return {
            "currency": "EUR",
            "period": {"start": start_d.isoformat() if start_d else None, "end": end_d.isoformat()},
            "income_cents": income,
            "expenses_cents": expenses,
            "net_cents": income - expenses,
            "excluded_transfers_cents": transfers,
            "unclassified_expenses_cents": unclassified,
            "by_category_cents": dict(sorted(by_cat.items(), key=lambda x: -x[1])),
            "by_bucket_cents": dict(sorted(by_bucket.items())),
            "transaction_count": len(transactions),
        }

    def _monthly_history(self, months: int = 3) -> dict[str, Any]:
        first_this_month = date(self.as_of.year, self.as_of.month, 1)
        start = add_months(first_this_month, -months)
        end = first_this_month - timedelta(days=1)
        filtered = self._tx(start, end)
        covered = sorted({t["date"].strftime("%Y-%m") for t in filtered})
        effective_months = max(1, len(covered))

        if not filtered:
            start = self.as_of - timedelta(days=29)
            end = self.as_of
            filtered = self._tx(start, end)
            covered = sorted({t["date"].strftime("%Y-%m") for t in filtered})
            effective_months = 1

        by_bucket: dict[str, int] = defaultdict(int)
        by_category: dict[str, int] = defaultdict(int)
        for tx in filtered:
            if tx["amount_cents"] < 0 and tx["bucket"] not in {"excluded_transfer", "income_reversal"}:
                by_bucket[tx["bucket"]] += -tx["amount_cents"]
                by_category[tx["category"]] += -tx["amount_cents"]
            elif tx["amount_cents"] > 0 and tx["bucket"] != "excluded_transfer":
                by_bucket["income"] += tx["amount_cents"]

        return {
            "months": effective_months,
            "start": start.isoformat(),
            "end": end.isoformat(),
            "months_with_transactions": covered,
            "by_bucket_average_cents": {k: proportional(v, 1, effective_months) for k, v in by_bucket.items()},
            "by_category_average_cents": {k: proportional(v, 1, effective_months) for k, v in by_category.items()},
        }

    def find_savings_opportunities(
        self,
        *,
        protected_categories: list[str] | None = None,
        months: int = 3,
        percentages: tuple[int, ...] = (10, 20),
    ) -> dict[str, Any]:
        protected = set(protected_categories or [])
        history = self._monthly_history(months)
        averages = history["by_category_average_cents"]
        opportunities = []
        for category, average in sorted(averages.items(), key=lambda x: -x[1]):
            if category not in SAVINGS_CANDIDATES or category in protected or average <= 0:
                continue
            options = []
            for pct in percentages:
                saved = proportional(average, pct, 100)
                options.append({
                    "reduction_percent": pct,
                    "monthly_saving_cents": saved,
                    "monthly_saving_eur": cents_to_eur(saved),
                    "annual_saving_cents": saved * 12,
                    "annual_saving_eur": cents_to_eur(saved * 12),
                })
            opportunities.append({
                "category": category,
                "category_fi": category_label_fi(category),
                "monthly_average_cents": average,
                "monthly_average_eur": cents_to_eur(average),
                "scenarios": options,
            })
        return {
            "currency": "EUR",
            "protected_categories": sorted(protected),
            "opportunities": opportunities,
        }

    def get_financial_forecast(
        self,
        *,
        monthly_everyday_budget_cents: int,
        monthly_savings_cents: int,
        months: int = 6,
        starting_checking_cents: int = 0,
        starting_buffer_cents: int = 0,
    ) -> list[dict[str, Any]]:
        records = []
        checking = starting_checking_cents
        buffer_bal = starting_buffer_cents
        goal_fund = 0
        half_save = proportional(monthly_savings_cents, 1, 2)
        other_half = monthly_savings_cents - half_save

        for idx in range(1, months + 1):
            buffer_bal += half_save
            goal_fund += other_half
            target_month = add_months(date(self.as_of.year, self.as_of.month, 1), idx).strftime("%Y-%m")
            records.append({
                "month": target_month,
                "checking_eur": cents_to_eur(checking),
                "buffer_eur": cents_to_eur(buffer_bal),
                "goal_fund_eur": cents_to_eur(goal_fund),
                "weekly_everyday_eur": cents_to_eur(monthly_to_weekly(monthly_everyday_budget_cents)),
            })
        return records

function escapeHtml(text) {
  return String(text || "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");

function fmtMonthsLabel(m) {
  const mo = t().moUnit;
  const yr = t().yrUnit;

  return `${m} ${mo}`;
}

function getCustomerPurchaseCategories(s) {
  const list = (s.everyday_categories_list || []).filter((x) => x.amount_eur > 0);
  if (list.length > 0) return list;
  return Object.entries(s.everyday_categories_30d || {}).map(([label, amount_eur]) => ({ label, amount_eur }));
}

function computeZeroSaveMonthlyEveryday() {
  const s = currentSnapshot;
  if (!s) return wizardState.everydayEur;
  const irreg = Number(s.baskets_30d?.irregular_reserve_eur || 35);
  const totalExp = wizardState.fixedEur + irreg + wizardState.everydayEur;
  const isDef = wizardState.incomeEur < totalExp;

  if (!isDef) {
    const netBeforeEv = wizardState.incomeEur - wizardState.fixedEur - irreg;
    return wizardState.everydayEur > 0 ? Math.min(wizardState.everydayEur, netBeforeEv) : netBeforeEv;
  }
  const bal = Number(s.account_balance_eur || 0);
  const resMo = Math.max(0, bal - Math.min(500, bal * 0.2)) / 12;
  const room = wizardState.incomeEur + resMo - wizardState.fixedEur - irreg;
  if (room > 0) return wizardState.everydayEur > 0 ? Math.min(wizardState.everydayEur, room) : room;
  return Math.max(60, Math.max(wizardState.incomeEur, bal / 6) * 0.18);
}

function getRecommendedMonthlySavings() {
  const zeroSaveEv = computeZeroSaveMonthlyEveryday();
  const spentWk = Number(currentSnapshot?.spent_this_week_eur || 0);
  const maxSafeMo = Math.max(0, Math.floor((zeroSaveEv - ((spentWk + 15) * 52) / 12) / 10) * 10);
  const gentle15Pct = Math.floor((zeroSaveEv * 0.15) / 10) * 10;
  return Math.min(150, maxSafeMo, Math.max(0, gentle15Pct));
}

function initWizardDefaults(s) {
  const p = s.profile || {};
  const b = s.baskets_30d || {};
  const userCats = getCustomerPurchaseCategories(s).map((c) => c.label);
  const defaultProtect = userCats.includes("Ruokakauppa") ? ["Ruokakauppa"] : userCats.slice(0, 1);
  const defaultCut = userCats.filter((c) => !defaultProtect.includes(c)).slice(0, 2);

  wizardState.incomeEur = Math.round(Number(s.monthly_income_eur || 0));
  wizardState.fixedEur = Math.round(Number(b.mandatory_regular_eur || 0));
  wizardState.everydayEur = Math.round(Number(s.average_everyday_monthly_eur || b.everyday_spent_eur || 0));
  wizardState.goal = p.active_goal?.name && GOAL_META.some((g) => g.id === p.active_goal.name) ? p.active_goal.name : "SÃ¤Ã¤stÃ¤Ã¤ tavoitteeseen";
  wizardState.cutFrom = p.onboarded && Array.isArray(p.cut_from) && p.cut_from.length ? [...p.cut_from] : defaultCut;
  wizardState.protect = p.onboarded && Array.isArray(p.priorities) && p.priorities.length ? [...p.priorities] : defaultProtect;
  wizardState.monthlySavings = p.onboarded ? Math.round(Number(b.monthly_savings_target_eur || 0)) : getRecommendedMonthlySavings();
  wizardState.targetAmount = p.active_goal?.target_eur || 1200;
}

function computeProjectedWeeklySafe(monthlySavingsVal) {
  const s = currentSnapshot;
  if (!s) return 0;
  const zeroSaveEv = computeZeroSaveMonthlyEveryday();
  const monthlyEvBudget = zeroSaveEv - monthlySavingsVal;
  return Math.round((monthlyEvBudget * 12) / 52 - Number(s.spent_this_week_eur || 0));
}

function selectSingleGoal(btnEl) {
  wizardState.goal = btnEl.dataset.goal;

function toggleCategoryPill(btnEl) {
  const cat = btnEl.dataset.cat;
  const group = btnEl.dataset.group;
  const otherGroup = group === "cutFrom" ? "protect" : "cutFrom";
  const otherContainerId = group === "cutFrom" ? "obProtectGrid" : "obCutGrid";
  const arr = wizardState[group];
  const idx = arr.indexOf(cat);
  if (idx >= 0) {
    arr.splice(idx, 1);
    btnEl.classList.remove("selected");
  } else {
    arr.push(cat);
    btnEl.classList.add("selected");
    const otherIdx = wizardState[otherGroup].indexOf(cat);
    if (otherIdx >= 0) {
      wizardState[otherGroup].splice(otherIdx, 1);
      document.querySelectorAll(`#${otherContainerId} .choice-pill`).forEach((b) => {
        if (b.dataset.cat === cat) b.classList.remove("selected");
      });
    }
  }
}

function renderHistory(history, showLoading = false) {
  const d = t();
  const lastMsg = history[history.length - 1];
  const isPlanJustConfirmed = Boolean(lastMsg && (lastMsg.status_type === "plan_confirmed" || lastMsg.selected_plan_id));

  const streams = [

  for (const { el, prefix, collapsed } of streams) {
    if (!el) continue;
    if (collapsed) {
      el.innerHTML = "";
      continue;
    }

    const messagesHtml = history.length
      ? history
          .map((m, idx) => {
            const isUser = m.role === "user";
            const cards = m.action_cards || [];

              : "";
            return `

                <div class="msg ${isUser ? "user" : "assistant"}">${formatMessageText(m.content)}</div>
                ${cardsHtml}
              </div>`;
          })
          .join("")
      : "";

    const doneBtnHtml =
      isPlanJustConfirmed && !showLoading
        ? prefix === "home"
          ? `<button type="button" class="chat-done-btn" onclick="hideMiniChat()">${d.doneHideChat}</button>`

  }

async function sendMessage(text) {
  const cleanText = String(text || "").trim();
  if (!cleanText || isSending || !currentCustomerId) return;

  homeChatCollapsed = false;
  const optimisticHistory = [...currentHistory, { role: "user", content: cleanText, action_cards: [] }];
  renderHistory(optimisticHistory, true);
  setSendingState(true);

  try {
    const res = await fetch("/api/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ statement_id: currentCustomerId, message: cleanText, lang: currentLang }),
    });
    if (!res.ok) throw new Error("Server error");
    const data = await res.json();
    renderSnapshot(data.updated_state);
    await loadState();
  } catch (err) {
    renderHistory([...optimisticHistory, { role: "assistant", content: `Error: ${err.message}`, action_cards: [] }]);
  } finally {
    setSendingState(false);
  }
}

  });

  renderSnapshot(currentSnapshot);
  renderHistory(currentHistory);
}
